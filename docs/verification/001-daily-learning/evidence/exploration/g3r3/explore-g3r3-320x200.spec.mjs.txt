/** 横スクロール、はみ出す要素、切れる文字（PRD AC-17 の幅 320 の確かめ方と同じ数え方）。 */
async function overflow(page) {
  return page.evaluate(() => {
    const d = document.documentElement;
    // 見た目だけ隠したファイルの input（contracts/identifiers.md の 2「例外は、見た目だけ隠したファイルの input（home.standalone.importFile、settings.import.file）」）は
    // 1 px に切り詰めてあるのが仕様なので、切れる文字の数え方から除く（1 回目の実行で S8 とダイアログ 2 つがこれで落ちた：探索テストの側の誤り）。
    const all = [...document.querySelectorAll('body *')].filter((el) => !(el.tagName === 'INPUT' && el.type === 'file'));
    const label = (el) => `${el.tagName.toLowerCase()}[data-testid=${el.getAttribute('data-testid')}]`;
    const wide = all.filter((el) => {
      const r = el.getBoundingClientRect();
      return r.width > 0 && (r.right > d.clientWidth + 1 || r.left < -1);
    }).map((el) => `${label(el)} left=${Math.round(el.getBoundingClientRect().left)} right=${Math.round(el.getBoundingClientRect().right)}`);
    const ellipsis = all.filter((el) => getComputedStyle(el).textOverflow === 'ellipsis' && ['hidden', 'clip'].includes(getComputedStyle(el).overflowX)).map(label);
    const clipped = all.filter((el) => {
      const cs = getComputedStyle(el);
      return ['hidden', 'clip'].includes(cs.overflowX) && el.scrollWidth > el.clientWidth + 1 && el.tagName !== 'DIALOG';
    }).map((el) => `${label(el)} scrollWidth=${el.scrollWidth} clientWidth=${el.clientWidth}`);
    return { scrollWidth: d.scrollWidth, clientWidth: d.clientWidth, scrollHeight: d.scrollHeight, wide, ellipsis, clipped,
      rootFontSize: getComputedStyle(d).fontSize };
  });
}
async function font200(page) {
  await page.evaluate(() => { document.documentElement.style.fontSize = '200%'; });
  await page.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r))));
}
/** 主な操作：件数、見えるか、大きさ、初期表示に入るか、スクロールすれば表示域の中に全部入るか。 */
async function ops(page, ids, { required = ids } = {}) {
  const out = [];
  for (const id of ids) {
    const loc = app.tid(page, id).first();
    const count = await app.tid(page, id).count();
    if (count === 0) { out.push({ id, count, required: required.includes(id) }); continue; }
    const before = await loc.boundingBox();
    const initiallyInView = !!before && before.y >= 0 && before.y + before.height <= VIEW.height && before.x >= 0 && before.x + before.width <= VIEW.width;
    await loc.scrollIntoViewIfNeeded();
    const b = await loc.boundingBox();
    const inViewAfterScroll = !!b && b.y >= -0.5 && b.y + b.height <= VIEW.height + 0.5 && b.x >= -0.5 && b.x + b.width <= VIEW.width + 0.5;
    const visible = await loc.isVisible();
    const text = (await loc.textContent())?.trim().replace(/\s+/g, ' ').slice(0, 40) ?? '';
    out.push({ id, count, required: required.includes(id), visible, text, box: b && { x: Math.round(b.x), y: Math.round(b.y), w: Math.round(b.width), h: Math.round(b.height) },
      initiallyInView, inViewAfterScroll, size44: !!b && b.width >= 44 && b.height >= 44 });
  }
  await page.evaluate(() => window.scrollTo(0, 0));
  return out;
}
async function record(page, testInfo, scene, { opIds, required, dialog, viewportOnly = false, extra = {} }) {
  const browser = testInfo.project.name;
  const o = await overflow(page);
  const operations = await ops(page, opIds, { required });
  const result = { scene, browser, viewport: `${VIEW.width}x${VIEW.height}`, state: await app.readState(page), ...o, operations, dialog: null, ...extra };
  if (dialog) {
    // design 9 と tools/screenshots.mjs の dialog の記録と同じ：初期フォーカスは「やめる」、移動キーで ok に届く。WebKit は Alt+Tab（test-hooks.md「キーボードの移動キー」）。
    const next = browser === 'webkit' ? 'Alt+Tab' : 'Tab';
    const activeId = () => page.evaluate(() => document.activeElement?.getAttribute('data-testid') ?? document.activeElement?.tagName);
    const rectOf = (id) => page.evaluate((i) => { const el = document.querySelector(`[data-testid="${i}"]`); if (!el) return null; const r = el.getBoundingClientRect();
      return { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height), inView: r.left >= 0 && r.right <= innerWidth && r.top >= 0 && r.bottom <= innerHeight }; }, id);
    const dialogRect = await rectOf(dialog.id);
    const initial = await activeId();
    const cancelRect = await rectOf(dialog.cancel);
    let okId = initial; let pressed = 0;
    for (; pressed < 6 && okId !== dialog.ok; pressed++) { await page.keyboard.press(next); okId = await activeId(); }
    const okRect = await rectOf(dialog.ok);
    result.dialog = { dialogRect, initialFocus: initial, cancelRect, keys: pressed, reached: okId, okRect };
  }
  mkdirSync(OUT, { recursive: true });
  await page.evaluate(() => window.scrollTo(0, 0));
  const png = path.join(OUT, `${browser}_${scene}.png`);
  await page.screenshot({ path: png, fullPage: !viewportOnly });
  writeFileSync(path.join(OUT, `${browser}_${scene}.json`), JSON.stringify(result, null, 1));
  await testInfo.attach(`EXPLORE_320x200_${scene}`, { body: JSON.stringify(result, null, 1), contentType: 'application/json' });
  // 判定
  expect(result.rootFontSize, 'ルートの文字サイズ 200%').toBe('32px');
  expect(result.scrollWidth, `${scene} の横スクロール（はみ出す要素 ${JSON.stringify(o.wide)}）`).toBe(result.clientWidth);
  expect(o.wide, `${scene} の表示域からはみ出す要素`).toEqual([]);
  expect(o.ellipsis, `${scene} の text-overflow: ellipsis`).toEqual([]);
  expect(o.clipped, `${scene} の overflow で切れた文字`).toEqual([]);
  for (const op of operations.filter((x) => x.required)) {
    expect(op.count, `${scene} の主な操作 ${op.id} の件数`).toBe(1);
    expect(op.visible, `${scene} の主な操作 ${op.id} が見える`).toBe(true);
    expect(op.inViewAfterScroll, `${scene} の主な操作 ${op.id} がスクロールで表示域に収まる ${JSON.stringify(op.box)}`).toBe(true);
    expect(op.size44, `${scene} の主な操作 ${op.id} が 44×44 以上 ${JSON.stringify(op.box)}`).toBe(true);
  }
  if (dialog) {
    expect(result.dialog.initialFocus, `${scene} の初期フォーカス`).toBe(dialog.cancel);
    expect(result.dialog.cancelRect?.inView, `${scene} の「やめる」が表示域の中 ${JSON.stringify(result.dialog.cancelRect)}`).toBe(true);
    expect(result.dialog.reached, `${scene} の移動キーで ok に届く（${result.dialog.keys} 回）`).toBe(dialog.ok);
    expect(result.dialog.okRect?.inView, `${scene} の ok が表示域の中 ${JSON.stringify(result.dialog.okRect)}`).toBe(true);
  }
  return result;
}
